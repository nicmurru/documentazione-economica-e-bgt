        // --- UTILS ---
        function parseIt(str) {
            if (!str) return 0;
            // Gestione robusta: toglie spazi, converte . in nulla e , in punto
            return parseFloat(str.toString().replace(/\s/g, '').replace(/\./g, '').replace(',', '.')) || 0;
        }
        function fmtMoney(n) {
            return '€ ' + n.toLocaleString('it-IT', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
        }

        // --- CALCOLO TOTALI (CRUSCOTTO) ---
        function updateDashboard(data) {
            // Usa un semplice ciclo for per massima velocità su grandi numeri
            let sBase = 0, sCec = 0, sRm = 0, sMano = 0, sBgt = 0;

            for (let i = 0; i < data.length; i++) {
                const row = data[i];
                sBase += parseIt(row[12]); // Col 12: Imp Base Gara
                sCec += parseIt(row[14]); // Col 14: Imp CEC
                sRm += parseIt(row[16]); // Col 16: Imp RM
                sMano += parseIt(row[19]); // Col 19: Imp Manodopera
                sBgt += parseIt(row[25]); // Col 25: Imp BGT
            }

            // Aggiorna DOM
            document.getElementById('dash-rows').innerText = data.length.toLocaleString('it-IT');
            document.getElementById('dash-base').innerText = fmtMoney(sBase);
            document.getElementById('dash-cec').innerText = fmtMoney(sCec);
            document.getElementById('dash-rm').innerText = fmtMoney(sRm);
            document.getElementById('dash-mano').innerText = fmtMoney(sMano);
            document.getElementById('dash-bgt').innerText = fmtMoney(sBgt);
        }

        // --- RENDER PAGINA ---
        function renderPage() {
            const tbody = document.getElementById('tableBody');
            const start = (currentPage - 1) * rowsPerPage;
            const end = start + rowsPerPage;
            const pageData = currentFilteredData.slice(start, end);

            let html = '';
            if (pageData.length === 0) {
                tbody.innerHTML = '<tr><td colspan="27" style="text-align:center; padding:20px;">Nessun dato trovato.</td></tr>';
                updatePaginationUI();
                return;
            }

            for (let i = 0; i < pageData.length; i++) {
                const row = pageData[i];
                html += '<tr onclick="selectRow(this)">'; // Aggiunto gestore riga
                for (let j = 0; j < row.length; j++) {
                    let cls = '';
                    let content = row[j] || '';
                    let onclickAttr = '';

                    if ((j >= 10 && j <= 19) || (j >= 24)) cls = 'num-cell';

                    // Colonna Descrizione (8)
                    if (j === 8) {
                        cls = 'desc-cell';
                        const safeContent = content.replace(/'/g, "\\'").replace(/"/g, '&quot;');
                        onclickAttr = `onclick="showDescription('${safeContent}', event)"`;
                        if (content.length > 70) content = content.substring(0, 70) + '...';
                    }

                    html += `<td class="${cls}" ${onclickAttr}>${content}</td>`;
                }
                html += '</tr>';
            }
            tbody.innerHTML = html;
            updatePaginationUI();
        }

        function updatePaginationUI() {
            const totalPages = Math.ceil(currentFilteredData.length / rowsPerPage) || 1;
            document.getElementById('pageInfo').innerText = `Pag. ${currentPage} / ${totalPages}`;
            document.getElementById('btnPrev').disabled = (currentPage === 1);
            document.getElementById('btnNext').disabled = (currentPage === totalPages);
        }

        function changePage(delta) {
            const totalPages = Math.ceil(currentFilteredData.length / rowsPerPage) || 1;
            const newPage = currentPage + delta;
            if (newPage >= 1 && newPage <= totalPages) {
                currentPage = newPage;
                renderPage();
                document.getElementById('tableContainer').scrollTop = 0;
            }
        }

                   <span class="btn-link" onclick="toggleAll(true)">Tutti</span>

        // Funzione principale per creare albero WBS - SMART NAMING
        function createWBSTree() {
            if (typeof tableData === 'undefined') {
                alert('Dati non ancora caricati. Riprova.');
                return;
            }

            const wbsData = {};
            window.wbsData = wbsData; // Expose globally for popup access
            const allWBS = new Set();

            // --- 1. COSTRUZIONE STRUTTURA E RAW DATA ---
            for (let i = 0; i < tableData.length; i++) {
                const row = tableData[i];
                const wbs = String(row[1] || 'NO_WBS');
                const importoRM = parseFloat(String(row[16] || '0').replace(/\./g, '').replace(',', '.')) || 0;
                const importoBGT = parseFloat(String(row[25] || '0').replace(/\./g, '').replace(',', '.')) || 0;

                allWBS.add(wbs);

                // Crea nodo se non esiste
                if (!wbsData[wbs]) {
                    wbsData[wbs] = {
                        code: wbs,
                        nome: '', // Sarà calcolato dopo
                        voci: 0,
                        importoRM: 0,
                        importoBGT: 0,
                        children: [],
                        rows: [] // Salviamo le righe grezze per analisi
                    };
                }

                // Accumula dati
                wbsData[wbs].voci++;
                wbsData[wbs].importoRM += importoRM;
                wbsData[wbs].importoBGT += importoBGT;
                wbsData[wbs].rows.push(row);
            }

            // Genera padri mancanti
            allWBS.forEach(function (wbs) {
                if (wbs.indexOf('.') > -1) {
                    const parts = wbs.split('.');
                    let parent = '';
                    for (let i = 0; i < parts.length - 1; i++) {
                        parent = parent ? parent + '.' + parts[i] : parts[i];
                        if (!wbsData[parent]) {
                            wbsData[parent] = {
                                code: parent,
                                nome: '',
                                voci: 0,
                                importoRM: 0,
                                importoBGT: 0,
                                children: [],
                                rows: []
                            };
                        }
                    }
                }
            });

            // Relazioni padre-figlio
            Object.keys(wbsData).forEach(function (code) {
                if (code.indexOf('.') > -1) {
                    const lastDot = code.lastIndexOf('.');
                    const parentCode = code.substring(0, lastDot);
                    if (wbsData[parentCode] && !wbsData[parentCode].children.includes(code)) {
                        wbsData[parentCode].children.push(code);
                    }
                }
            });

            // --- 2. POPOLAMENTO RIGHE RICORSIVO (per i padri) ---
            // I padri devono conoscere le righe dei figli per capire "cosa contengono"
            function aggregateRows(nodeCode) {
                const node = wbsData[nodeCode];
                // Se ha figli, aggrega le loro righe
                if (node.children.length > 0) {
                    let sumRM = 0;
                    let sumBGT = 0;
                    node.children.forEach(childCode => {
                        const childNode = aggregateRows(childCode);
                        // Propaga le righe su
                        node.rows = node.rows.concat(childNode.rows);

                        // Somma importi dai figli
                        sumRM += childNode.importoRM;
                        sumBGT += childNode.importoBGT;
                    });

                    // Aggiungi i totali dei figli ai totali del nodo corrente (che ha già i suoi importi diretti)
                    node.importoRM += sumRM;
                    node.importoBGT += sumBGT;
                }
                return node;
            }

            // Lancia aggregazione dalle radici
            const rootNodes = Object.keys(wbsData).filter(c => c.indexOf('.') === -1);
            rootNodes.forEach(root => aggregateRows(root));


            // --- 3. INTELLIGENT NAMING ---
            // Definizione Colonne Gerarchia
            // 3: Gruppo, 4: Super Cat, 5: Categorie, 6: Sub Cat
            // 22: Tipologia, 23: Sottotipologia
            const hierarchyCols = [3, 4, 5, 6];
            const extraCols = [22, 23];

            function getCommonValue(rows, colIdx) {
                if (!rows || rows.length === 0) return '';
                const firstVal = rows[0][colIdx] || '';
                for (let i = 1; i < rows.length; i++) {
                    if ((rows[i][colIdx] || '') !== firstVal) return null; // Mixed
                }
                return firstVal;
            }

            function getMostFrequentValue(rows, colIdx) {
                if (!rows || rows.length === 0) return '';
                const counts = {};
                let maxCount = 0;
                let bestVal = '';

                for (let i = 0; i < rows.length; i++) {
                    const val = (rows[i][colIdx] || '').trim();
                    if (!val) continue;

                    counts[val] = (counts[val] || 0) + 1;
                    if (counts[val] > maxCount) {
                        maxCount = counts[val];
                        bestVal = val;
                    }
                }
                return bestVal;
            }

            function getBestNameForNode(node, levelIndex) {
                // 0. OVERRIDES MANUALI (Richiesti utente)
                const manualNames = {
                    '12': 'Bonifica ordigni bellici',
                    '13': 'Bilancio terre'
                };
                if (manualNames[node.code]) return manualNames[node.code];

                // 1. Logica Standard (Livello gerarchico)
                let colIdx = hierarchyCols[Math.min(levelIndex, hierarchyCols.length - 1)];
                let name = getCommonValue(node.rows, colIdx);

                // 2. Se non c'è unanimità (Misto), usa il valore più frequente (es. per il nodo 7)
                // Invece di scendere "Deepest", restiamo sul livello ma prendiamo la maggioranza
                if (!name) name = getMostFrequentValue(node.rows, colIdx);

                // 3. Se ancora vuoto (es. livello senza dati), prova a scendere nelle colonne successive
                if (!name) {
                    for (let k = levelIndex + 1; k < hierarchyCols.length; k++) {
                        const fallbackCol = hierarchyCols[k];
                        const fallback = getCommonValue(node.rows, fallbackCol) || getMostFrequentValue(node.rows, fallbackCol);
                        if (fallback) {
                            name = fallback;
                            break;
                        }
                    }
                }

                // 4. Ultimo tentativo con Tipologia(22)
                if (!name) name = getCommonValue(node.rows, 22) || getMostFrequentValue(node.rows, 22);

                // 5. Default
                return name || ('Nodo ' + node.code);
            }

            // Assegna nomi base
            Object.values(wbsData).forEach(node => {
                const level = node.code.split('.').length - 1; // 0-based level
                node.nome = getBestNameForNode(node, level);

            });

            // --- 4. RISOLUZIONE CONFLITTI (Disambiguazione) ---
            // Se due fratelli hanno lo stesso nome, bisogna aggiungere dettagli
            function resolveConflicts(nodesList) {
                // Raggruppa per nome
                const nameMap = {};
                nodesList.forEach(n => {
                    if (!nameMap[n.nome]) nameMap[n.nome] = [];
                    nameMap[n.nome].push(n);
                });

                // Itera sui gruppi con collisioni
                Object.keys(nameMap).forEach(name => {
                    const group = nameMap[name];
                    if (group.length > 1) {
                        // C'è conflitto. Cerchiamo un discriminante.
                        // Proviamo colonne extra in ordine: Categorie(5), SubCat(6), Tipologia(22), Sottotipologia(23)
                        // Escludendo quella usata per il nome base (assumiamo euristica)
                        const candidates = [5, 6, 22, 23];

                        for (let col of candidates) {
                            // Verifica se questa colonna offre valori diversi per i nodi del gruppo
                            const values = new Set();
                            group.forEach(n => {
                                const val = getCommonValue(n.rows, col) || 'Misto';
                                values.add(val);
                            });

                            if (values.size > 1) {
                                // Trovato discriminante! Aggiorna nomi
                                group.forEach(n => {
                                    const val = getCommonValue(n.rows, col); // Può essere null se 'Misto' all'interno del nodo stesso
                                    if (val && val !== n.nome) {
                                        n.nome += ' - ' + val;
                                    } else {
                                        // Se anche interno è misto, o uguale al nome, non aiuta molto, ma proviamo
                                        // Fallback estremo: aggiungi codice
                                        n.nome += ' (' + n.code + ')';
                                    }
                                });
                                return; // Risolto per questo gruppo (si spera)
                            }
                        }

                        // Se nessun discriminante trovato, aggiungi codice
                        group.forEach(n => {
                            n.nome += ' (' + n.code + ')';
                        });
                    }
                });
            }

            // Applica risoluzione conflitti a ogni livello di "siblings"
            // 1. Radici
            resolveConflicts(rootNodes.map(k => wbsData[k]));
            // 2. Figli di ogni nodo
            Object.values(wbsData).forEach(node => {
                if (node.children.length > 0) {
                    resolveConflicts(node.children.map(k => wbsData[k]));
                }
            });


            // --- 5. RENDER (Invariato ma pulito) ---

            if (!newWin) { alert('Popup bloccato. Abilita i popup.'); return; }

            function writeNode(code) {
                const node = wbsData[code];
                if (!node) return;
                const hasChildren = node.children.length > 0;

                newWin.document.write('<div class="node-wrapper">');
                newWin.document.write('<div class="wbs-row">');

                // Toggler
                if (hasChildren) {
                    newWin.document.write('<span class="toggler" onclick="toggle(this)">▼</span>');
                } else {
                    newWin.document.write('<span class="toggler" style="opacity:0.3">•</span>');
                }

                newWin.document.write('<div class="wbs-content">');
                newWin.document.write('<span class="wbs-code">' + node.code + '</span>');
                newWin.document.write('<span class="wbs-name" onclick="window.opener.showNodeDetails(\'' + node.code + '\', this)" style="cursor:pointer; text-decoration:underline; color:#0056b3;" title="Clicca per vedere le voci">' + node.nome + '</span>');

                newWin.document.write('<div class="wbs-meta">');
                newWin.document.write('<span>' + node.voci + ' voci</span>');
                newWin.document.write('<span class="wbs-currency" style="color:#003366">RM ' + node.importoRM.toLocaleString('it-IT', { minimumFractionDigits: 0 }) + '</span>');
                newWin.document.write('<span class="wbs-currency" style="color:#e67e22">BGT ' + node.importoBGT.toLocaleString('it-IT', { minimumFractionDigits: 0 }) + '</span>');
                newWin.document.write('</div>'); // end meta

                newWin.document.write('</div>'); // end content
                newWin.document.write('</div>'); // end row

                if (hasChildren) {
                    newWin.document.write('<div class="children-container">');
                    node.children.sort((a, b) => {
                        // Sort numerico semantico (1.2 vs 1.10)
                        return a.localeCompare(b, undefined, { numeric: true, sensitivity: 'base' });
                    }).forEach(child => writeNode(child));

            html += '<th>Importo RM</th>';
            html += '</tr></thead><tbody>';

        // --- EXPORT FUNCTIONS ---

        function copyToClipboard(text) {
            if (navigator.clipboard) {
                navigator.clipboard.writeText(text).then(() => {
                    alert("Dati copiati negli appunti. Incolla in Excel.");
                }).catch(err => {
                    console.error('Errore copia appunti: ', err);
                    fallbackCopyTextToClipboard(text);
                });
            } else {
                fallbackCopyTextToClipboard(text);
            }
        }

        function fallbackCopyTextToClipboard(text) {
            var textArea = document.createElement("textarea");
            textArea.value = text;

            // Ensure element is visible but out of viewport
            textArea.style.position = "fixed";
            textArea.style.left = "-9999px";
            textArea.style.top = "0";

            document.body.appendChild(textArea);
            textArea.focus();
            textArea.select();

            try {
                var successful = document.execCommand('copy');
                var msg = successful ? 'Dati copiati negli appunti. Incolla in Excel.' : 'Copia fallita';
                alert(msg);
            } catch (err) {
                console.error('Baco Copia fallback', err);
                alert('Impossibile copiare automaticamente. Riprova.');
            }

            document.body.removeChild(textArea);
        }

        function exportDataToExcel(data, headers) {
            let tsv = headers.join('\t') + '\n';
            data.forEach(row => {
                let line = row.map(cell => {
                    let txt = String(cell || '').replace(/(\r\n|\n|\r)/gm, " "); // Rimuovi newline interni
                    txt = txt.replace(/\t/g, " "); // Rimuovi tab interni
                    return txt;
                }).join('\t');
                tsv += line + '\n';
            });
            copyToClipboard(tsv);
        }

        async function checkPassword() {
            const input = document.getElementById('password-input');
            const error = document.getElementById('password-error');
            const val = input.value;

            // Calculate Hash
            const msgBuffer = new TextEncoder().encode(val);
            const hashBuffer = await crypto.subtle.digest('SHA-256', msgBuffer);
            const hashArray = Array.from(new Uint8Array(hashBuffer));
            const hashHex = hashArray.map(b => b.toString(16).padStart(2, '0')).join('');

            // TARGET HASH = SHA256('RM_@lghero_2025')
            const targetHash = "3075ef1f7aa38bfccd3ebb4987c36d4f86f223cd3bdbd7689c1a3f2873f3aa40";

            if (hashHex === targetHash) {
                document.getElementById('password-overlay').style.display = 'none';
                document.getElementById('landing-page').style.display = 'flex';
            } else {
                error.style.display = 'block';
                input.style.borderColor = '#d32f2f';
            }
        }

        // Enter key support
        document.getElementById('password-input').addEventListener('keypress', function (e) {
            if (e.key === 'Enter') {
